
This uses GCP network tags for firewall targeting, which is the correct way to apply firewall rules to selected VMs. Google Cloud docs state that target tags make firewall rules applicable to matching VM instances. ([Google Cloud Documentation][1])
The Nagios install is based on Ubuntu’s `nagios4` package approach, and the web fix uses Apache 2.4 authorization with `Require valid-user`. ([Ubuntu Community Hub][2])

---

# Full working script — run in Cloud Shell

```bash
cat > nagios_gcp_full_setup.sh <<'EOF'
#!/usr/bin/env bash
set -euo pipefail

# =========================
# User-editable variables
# =========================
SERVER_NAME="${SERVER_NAME:-nagios-server}"
CLIENT_NAME="${CLIENT_NAME:-gcp-client-1}"

DEFAULT_ZONE="${DEFAULT_ZONE:-asia-south1-a}"
SERVER_MACHINE_TYPE="${SERVER_MACHINE_TYPE:-e2-medium}"
CLIENT_MACHINE_TYPE="${CLIENT_MACHINE_TYPE:-e2-micro}"

IMAGE_FAMILY="${IMAGE_FAMILY:-ubuntu-2204-lts}"
IMAGE_PROJECT="${IMAGE_PROJECT:-ubuntu-os-cloud}"

SERVER_TAG="${SERVER_TAG:-nagios-server}"
CLIENT_TAG="${CLIENT_TAG:-nagios-client}"

NAGIOS_USER="${NAGIOS_USER:-nagiosadmin}"
NAGIOS_PASSWORD="${NAGIOS_PASSWORD:-Nagios@123}"

FW_HTTP_RULE="${FW_HTTP_RULE:-allow-nagios-http}"
FW_CLIENT_RULE="${FW_CLIENT_RULE:-allow-nagios-to-client}"

# For lab: 0.0.0.0/0.
# Later restrict this to your own public IP/32.
HTTP_SOURCE_RANGE="${HTTP_SOURCE_RANGE:-0.0.0.0/0}"

# =========================
# Project check
# =========================
PROJECT_ID="$(gcloud config get-value project 2>/dev/null || true)"

if [[ -z "$PROJECT_ID" ]]; then
  echo "ERROR: No GCP project configured."
  echo "Run: gcloud config set project YOUR_PROJECT_ID"
  exit 1
fi

echo "Using project: $PROJECT_ID"

# =========================
# Helper functions
# =========================
instance_zone() {
  local name="$1"
  gcloud compute instances list \
    --filter="name=($name)" \
    --format="value(zone.basename())" \
    | head -n 1
}

instance_exists() {
  local name="$1"
  local zone
  zone="$(instance_zone "$name")"
  [[ -n "$zone" ]]
}

wait_for_ssh() {
  local name="$1"
  local zone="$2"

  echo "Waiting for SSH on $name..."
  for i in {1..30}; do
    if gcloud compute ssh "$name" \
      --zone="$zone" \
      --command="echo SSH_OK" \
      --quiet >/dev/null 2>&1; then
      echo "SSH is ready on $name"
      return 0
    fi
    echo "SSH not ready yet for $name. Retry $i/30..."
    sleep 10
  done

  echo "ERROR: SSH did not become ready for $name"
  exit 1
}

# =========================
# Create VMs if missing
# =========================
SERVER_ZONE="$(instance_zone "$SERVER_NAME")"

if [[ -z "$SERVER_ZONE" ]]; then
  SERVER_ZONE="$DEFAULT_ZONE"
  echo "Creating Nagios server VM: $SERVER_NAME in $SERVER_ZONE"

  gcloud compute instances create "$SERVER_NAME" \
    --zone="$SERVER_ZONE" \
    --machine-type="$SERVER_MACHINE_TYPE" \
    --image-family="$IMAGE_FAMILY" \
    --image-project="$IMAGE_PROJECT" \
    --boot-disk-size=20GB \
    --tags="$SERVER_TAG"
else
  echo "Nagios server VM already exists: $SERVER_NAME in $SERVER_ZONE"
  gcloud compute instances add-tags "$SERVER_NAME" \
    --zone="$SERVER_ZONE" \
    --tags="$SERVER_TAG" >/dev/null
fi

CLIENT_ZONE="$(instance_zone "$CLIENT_NAME")"

if [[ -z "$CLIENT_ZONE" ]]; then
  CLIENT_ZONE="$DEFAULT_ZONE"
  echo "Creating Nagios client VM: $CLIENT_NAME in $CLIENT_ZONE"

  gcloud compute instances create "$CLIENT_NAME" \
    --zone="$CLIENT_ZONE" \
    --machine-type="$CLIENT_MACHINE_TYPE" \
    --image-family="$IMAGE_FAMILY" \
    --image-project="$IMAGE_PROJECT" \
    --boot-disk-size=10GB \
    --tags="$CLIENT_TAG"
else
  echo "Nagios client VM already exists: $CLIENT_NAME in $CLIENT_ZONE"
  gcloud compute instances add-tags "$CLIENT_NAME" \
    --zone="$CLIENT_ZONE" \
    --tags="$CLIENT_TAG" >/dev/null
fi

wait_for_ssh "$SERVER_NAME" "$SERVER_ZONE"
wait_for_ssh "$CLIENT_NAME" "$CLIENT_ZONE"

# =========================
# Dynamic values
# =========================
SERVER_INTERNAL_IP="$(gcloud compute instances describe "$SERVER_NAME" \
  --zone="$SERVER_ZONE" \
  --format="value(networkInterfaces[0].networkIP)")"

SERVER_EXTERNAL_IP="$(gcloud compute instances describe "$SERVER_NAME" \
  --zone="$SERVER_ZONE" \
  --format="value(networkInterfaces[0].accessConfigs[0].natIP)")"

CLIENT_INTERNAL_IP="$(gcloud compute instances describe "$CLIENT_NAME" \
  --zone="$CLIENT_ZONE" \
  --format="value(networkInterfaces[0].networkIP)")"

SERVER_NETWORK="$(gcloud compute instances describe "$SERVER_NAME" \
  --zone="$SERVER_ZONE" \
  --format="value(networkInterfaces[0].network.basename())")"

CLIENT_NETWORK="$(gcloud compute instances describe "$CLIENT_NAME" \
  --zone="$CLIENT_ZONE" \
  --format="value(networkInterfaces[0].network.basename())")"

if [[ "$SERVER_NETWORK" != "$CLIENT_NETWORK" ]]; then
  echo "ERROR: Server and client are in different VPC networks."
  echo "Server network: $SERVER_NETWORK"
  echo "Client network: $CLIENT_NETWORK"
  exit 1
fi

if [[ -z "$SERVER_EXTERNAL_IP" ]]; then
  echo "ERROR: $SERVER_NAME does not have an external IP."
  echo "Add an external IP or use IAP/SSH tunnel."
  exit 1
fi

echo
echo "Dynamic values discovered:"
echo "SERVER_NAME=$SERVER_NAME"
echo "CLIENT_NAME=$CLIENT_NAME"
echo "SERVER_ZONE=$SERVER_ZONE"
echo "CLIENT_ZONE=$CLIENT_ZONE"
echo "SERVER_INTERNAL_IP=$SERVER_INTERNAL_IP"
echo "SERVER_EXTERNAL_IP=$SERVER_EXTERNAL_IP"
echo "CLIENT_INTERNAL_IP=$CLIENT_INTERNAL_IP"
echo "NETWORK=$SERVER_NETWORK"
echo

# =========================
# Firewall rules
# =========================
echo "Recreating firewall rules..."

gcloud compute firewall-rules delete "$FW_HTTP_RULE" --quiet >/dev/null 2>&1 || true

gcloud compute firewall-rules create "$FW_HTTP_RULE" \
  --network="$SERVER_NETWORK" \
  --allow=tcp:80 \
  --source-ranges="$HTTP_SOURCE_RANGE" \
  --target-tags="$SERVER_TAG"

gcloud compute firewall-rules delete "$FW_CLIENT_RULE" --quiet >/dev/null 2>&1 || true

gcloud compute firewall-rules create "$FW_CLIENT_RULE" \
  --network="$SERVER_NETWORK" \
  --allow=icmp,tcp:22,tcp:80,tcp:5666 \
  --source-tags="$SERVER_TAG" \
  --target-tags="$CLIENT_TAG"

# =========================
# Configure client VM
# =========================
echo
echo "Configuring Nagios client: $CLIENT_NAME"

gcloud compute ssh "$CLIENT_NAME" \
  --zone="$CLIENT_ZONE" \
  --quiet \
  --command="sudo bash -s -- '$SERVER_INTERNAL_IP'" <<'CLIENT_SCRIPT'
set -euo pipefail

SERVER_INTERNAL_IP="$1"
export DEBIAN_FRONTEND=noninteractive

echo "Installing NRPE, monitoring plugins, and nginx..."
apt-get update
apt-get install -y nagios-nrpe-server monitoring-plugins nginx net-tools curl

echo "Configuring NRPE allowed_hosts..."
cp -a /etc/nagios/nrpe.cfg /etc/nagios/nrpe.cfg.backup.$(date +%F-%H%M%S)

if grep -q '^allowed_hosts=' /etc/nagios/nrpe.cfg; then
  sed -i "s/^allowed_hosts=.*/allowed_hosts=127.0.0.1,::1,${SERVER_INTERNAL_IP}/" /etc/nagios/nrpe.cfg
else
  echo "allowed_hosts=127.0.0.1,::1,${SERVER_INTERNAL_IP}" >> /etc/nagios/nrpe.cfg
fi

# Avoid localhost-only binding if present.
sed -i 's/^server_address=.*/#server_address=/' /etc/nagios/nrpe.cfg || true

echo "Adding NRPE commands..."
sed -i '/^command\[check_root_disk\]/d' /etc/nagios/nrpe.cfg
sed -i '/^command\[check_load\]/d' /etc/nagios/nrpe.cfg
sed -i '/^command\[check_total_procs\]/d' /etc/nagios/nrpe.cfg

cat >> /etc/nagios/nrpe.cfg <<'EOF_NRPE'

command[check_root_disk]=/usr/lib/nagios/plugins/check_disk -w 20% -c 10% -p /
command[check_load]=/usr/lib/nagios/plugins/check_load -w 5,4,3 -c 10,6,4
command[check_total_procs]=/usr/lib/nagios/plugins/check_procs -w 150 -c 200
EOF_NRPE

echo "Starting services..."
systemctl enable nagios-nrpe-server
systemctl restart nagios-nrpe-server

systemctl enable nginx
systemctl restart nginx

echo "Client listening ports:"
ss -lntp | grep -E ':80|:5666' || true

echo "Client configuration completed."
CLIENT_SCRIPT

# =========================
# Configure Nagios server VM
# =========================
echo
echo "Configuring Nagios server: $SERVER_NAME"

gcloud compute ssh "$SERVER_NAME" \
  --zone="$SERVER_ZONE" \
  --quiet \
  --command="sudo bash -s -- '$NAGIOS_USER' '$NAGIOS_PASSWORD' '$CLIENT_NAME' '$CLIENT_INTERNAL_IP'" <<'SERVER_SCRIPT'
set -euo pipefail

NAGIOS_USER="$1"
NAGIOS_PASSWORD="$2"
CLIENT_NAME="$3"
CLIENT_INTERNAL_IP="$4"

export DEBIAN_FRONTEND=noninteractive

echo "Preseeding postfix to avoid interactive prompt..."
echo "postfix postfix/main_mailer_type select No configuration" | debconf-set-selections || true

echo "Installing Nagios, Apache, plugins, and NRPE plugin..."
apt-get update
apt-get install -y apache2 apache2-utils nagios4 nagios-nrpe-plugin monitoring-plugins curl net-tools

echo "Enabling Apache modules..."
a2enmod cgi rewrite auth_basic authn_file authz_user authz_core authz_host dir >/dev/null

echo "Backing up Apache/Nagios config..."
BACKUP_DIR="/root/nagios-working-backup-$(date +%F-%H%M%S)"
mkdir -p "$BACKUP_DIR"
cp -a /etc/apache2 "$BACKUP_DIR/apache2" || true
cp -a /etc/nagios4 "$BACKUP_DIR/nagios4" || true

echo "Removing conflicting Nagios Apache configs..."
rm -f /etc/apache2/conf-enabled/nagios4*.conf
rm -f /etc/apache2/sites-enabled/nagios4*.conf

echo "Creating Nagios web login user..."
htpasswd -bc /etc/nagios4/htpasswd.users "$NAGIOS_USER" "$NAGIOS_PASSWORD"
chown root:www-data /etc/nagios4/htpasswd.users
chmod 640 /etc/nagios4/htpasswd.users

echo "Fixing Nagios web directory permissions..."
chmod 755 /usr/share/nagios4 || true
chmod 755 /usr/share/nagios4/htdocs || true
chmod 755 /usr/lib/cgi-bin/nagios4 || true
chmod 755 /etc/nagios4 || true
chmod 755 /etc/nagios4/stylesheets 2>/dev/null || true

echo "Creating safe Nagios index redirect..."
cat > /usr/share/nagios4/htdocs/index.html <<'EOF_INDEX'
<html>
<head>
<meta http-equiv="refresh" content="0;url=/nagios4/cgi-bin/status.cgi?host=all">
</head>
<body>
<a href="/nagios4/cgi-bin/status.cgi?host=all">Open Nagios</a>
</body>
</html>
EOF_INDEX

echo "Creating clean Apache config for Nagios..."
cat > /etc/apache2/conf-available/nagios4-fixed.conf <<'EOF_APACHE'
ScriptAlias /nagios4/cgi-bin /usr/lib/cgi-bin/nagios4
ScriptAlias /cgi-bin/nagios4 /usr/lib/cgi-bin/nagios4

Alias /nagios4/stylesheets /etc/nagios4/stylesheets
Alias /nagios4 /usr/share/nagios4/htdocs

<Directory "/usr/share/nagios4/htdocs">
    Options FollowSymLinks
    DirectoryIndex index.html index.php
    AllowOverride None

    AuthType Basic
    AuthName "Nagios Access"
    AuthBasicProvider file
    AuthUserFile /etc/nagios4/htpasswd.users
    Require valid-user
</Directory>

<Directory "/usr/lib/cgi-bin/nagios4">
    Options ExecCGI FollowSymLinks
    AllowOverride None
    AddHandler cgi-script .cgi

    AuthType Basic
    AuthName "Nagios Access"
    AuthBasicProvider file
    AuthUserFile /etc/nagios4/htpasswd.users
    Require valid-user
</Directory>

<Directory "/etc/nagios4/stylesheets">
    Options FollowSymLinks
    AllowOverride None

    AuthType Basic
    AuthName "Nagios Access"
    AuthBasicProvider file
    AuthUserFile /etc/nagios4/htpasswd.users
    Require valid-user
</Directory>
EOF_APACHE

a2enconf nagios4-fixed >/dev/null

echo "Fixing Nagios CGI authentication/authorization..."
sed -i 's/^use_authentication=.*/use_authentication=1/' /etc/nagios4/cgi.cfg

for key in \
authorized_for_system_information \
authorized_for_configuration_information \
authorized_for_system_commands \
authorized_for_all_services \
authorized_for_all_hosts \
authorized_for_all_service_commands \
authorized_for_all_host_commands
do
  if grep -q "^${key}=" /etc/nagios4/cgi.cfg; then
    sed -i "s/^${key}=.*/${key}=${NAGIOS_USER}/" /etc/nagios4/cgi.cfg
  else
    echo "${key}=${NAGIOS_USER}" >> /etc/nagios4/cgi.cfg
  fi
done

echo "Creating NRPE command definition..."
cat > /etc/nagios4/conf.d/check_nrpe_dynamic_command.cfg <<'EOF_NRPE_COMMAND'
define command {
    command_name    check_nrpe_dynamic
    command_line    /usr/lib/nagios/plugins/check_nrpe -H $HOSTADDRESS$ -c $ARG1$
}
EOF_NRPE_COMMAND

echo "Creating Nagios host/service config for client..."
cat > "/etc/nagios4/conf.d/${CLIENT_NAME}.cfg" <<EOF_CLIENT_CFG
define host {
    use                     generic-host
    host_name               ${CLIENT_NAME}
    alias                   GCP Client VM
    address                 ${CLIENT_INTERNAL_IP}
    max_check_attempts      5
    check_period            24x7
    notification_interval   30
    notification_period     24x7
}

define service {
    use                     generic-service
    host_name               ${CLIENT_NAME}
    service_description     PING
    check_command           check-host-alive
}

define service {
    use                     generic-service
    host_name               ${CLIENT_NAME}
    service_description     SSH
    check_command           check_ssh
}

define service {
    use                     generic-service
    host_name               ${CLIENT_NAME}
    service_description     HTTP
    check_command           check_http
}

define service {
    use                     generic-service
    host_name               ${CLIENT_NAME}
    service_description     Root Disk
    check_command           check_nrpe_dynamic!check_root_disk
}

define service {
    use                     generic-service
    host_name               ${CLIENT_NAME}
    service_description     Current Load
    check_command           check_nrpe_dynamic!check_load
}

define service {
    use                     generic-service
    host_name               ${CLIENT_NAME}
    service_description     Total Processes
    check_command           check_nrpe_dynamic!check_total_procs
}
EOF_CLIENT_CFG

echo "Validating Apache config..."
apache2ctl configtest

echo "Validating Nagios config..."
nagios4 -v /etc/nagios4/nagios.cfg

echo "Restarting services..."
systemctl enable apache2
systemctl enable nagios4
systemctl restart apache2
systemctl restart nagios4

echo "Testing local Nagios URL without auth. 401 is expected:"
curl -I http://127.0.0.1/nagios4/ || true

echo "Testing local Nagios CGI with auth:"
curl -I -u "${NAGIOS_USER}:${NAGIOS_PASSWORD}" "http://127.0.0.1/nagios4/cgi-bin/status.cgi?host=all" || true

echo "Testing Nagios service status:"
systemctl --no-pager status apache2 | head -n 20 || true
systemctl --no-pager status nagios4 | head -n 20 || true

echo "Nagios server configuration completed."
SERVER_SCRIPT

# =========================
# Final tests from Nagios server
# =========================
echo
echo "Running final tests from Nagios server..."

gcloud compute ssh "$SERVER_NAME" \
  --zone="$SERVER_ZONE" \
  --quiet \
  --command="
set -e

echo 'Testing /nagios4 locally. HTTP 401 means auth is working:'
curl -I http://127.0.0.1/nagios4/ || true

echo
echo 'Testing NRPE version from Nagios server to client:'
/usr/lib/nagios/plugins/check_nrpe -H '$CLIENT_INTERNAL_IP' || true

echo
echo 'Testing root disk via NRPE:'
/usr/lib/nagios/plugins/check_nrpe -H '$CLIENT_INTERNAL_IP' -c check_root_disk || true

echo
echo 'Testing load via NRPE:'
/usr/lib/nagios/plugins/check_nrpe -H '$CLIENT_INTERNAL_IP' -c check_load || true

echo
echo 'Testing process count via NRPE:'
/usr/lib/nagios/plugins/check_nrpe -H '$CLIENT_INTERNAL_IP' -c check_total_procs || true

echo
echo 'Testing HTTP on client:'
/usr/lib/nagios/plugins/check_http -H '$CLIENT_INTERNAL_IP' || true

echo
echo 'Final Nagios config validation:'
sudo nagios4 -v /etc/nagios4/nagios.cfg | tail -n 20
"

echo
echo "=================================================="
echo "SETUP COMPLETED"
echo "=================================================="
echo "Nagios URL      : http://${SERVER_EXTERNAL_IP}/nagios4/"
echo "Username        : ${NAGIOS_USER}"
echo "Password        : ${NAGIOS_PASSWORD}"
echo
echo "Server VM       : ${SERVER_NAME}"
echo "Server zone     : ${SERVER_ZONE}"
echo "Server private  : ${SERVER_INTERNAL_IP}"
echo "Server public   : ${SERVER_EXTERNAL_IP}"
echo
echo "Client VM       : ${CLIENT_NAME}"
echo "Client zone     : ${CLIENT_ZONE}"
echo "Client private  : ${CLIENT_INTERNAL_IP}"
echo
echo "Open:"
echo "http://${SERVER_EXTERNAL_IP}/nagios4/"
echo
echo "If page asks for login, it is working."
echo "If local curl shows 401 Unauthorized, that is also correct."
echo "=================================================="
EOF

chmod +x nagios_gcp_full_setup.sh
./nagios_gcp_full_setup.sh
```

# What this script fixes

It fixes your earlier `Forbidden` issue by removing conflicting Apache Nagios configs and creating one clean config:

```text
/etc/apache2/conf-available/nagios4-fixed.conf
```

The key working rule is:

```apache
Require valid-user
```

Earlier, your config was likely denying external access to `/nagios4` before login. Correct behavior is:

```text
403 Forbidden        = wrong, Apache denied access
401 Unauthorized    = correct, Apache is asking for login
200 OK after login   = correct, Nagios page loaded
```

# Security note

The script opens Nagios UI to:

```text
0.0.0.0/0
```

For a lab this is okay, but later restrict it to your public IP:

```bash
export MY_PUBLIC_IP="YOUR_PUBLIC_IP/32"

gcloud compute firewall-rules update allow-nagios-http \
  --source-ranges="$MY_PUBLIC_IP"
```

Also change the default password after testing:

```bash
gcloud compute ssh nagios-server --zone=asia-south1-a
sudo htpasswd /etc/nagios4/htpasswd.users nagiosadmin
sudo systemctl restart apache2
```
